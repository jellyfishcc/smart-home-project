#include <Arduino.h>
#include <U8g2lib.h>
#include <Wire.h>
#include <DHT.h>
#include <Adafruit_NeoPixel.h>


//================ OLED =================

U8G2_SH1106_128X64_NONAME_1_4W_HW_SPI oled(
  U8G2_R0,
  10,
  9,
  8
);


//================ DHT11 ================

#define DHT_PIN 7

#define DHT_TYPE DHT11

DHT dht(
  DHT_PIN,
  DHT_TYPE
);


//================ FAN =================

#define FAN_PIN 5



//================ WS2812 ===============

#define LED_PIN 4

#define LED_NUM 8


Adafruit_NeoPixel strip(
  LED_NUM,
  LED_PIN,
  NEO_GRB + NEO_KHZ800
);



//================ 光敏DO ===============

#define LDR_PIN 6



//================ 参数 =================

// 风扇开启温度
#define FAN_ON_TEMP 30

// 风扇关闭温度
#define FAN_OFF_TEMP 28



//================ 数据变量 =============

float temperature=25;
bool fanState=false;
float humidity=50;


bool darkState=false;

bool oldDarkState=false;

int lightCount=0;

bool lastLight=false;




unsigned long dhtTimer=0;

unsigned long oledTimer=0;
float oldTemp=-100;
float oldHum=-100;
bool oldFan=false;


//================ 串口回传 QENV（新增） ================
//
// 契约：收到 "QENV\n" -> 回 "@DATA ENV <温度x10> <湿度x10> <DHT有效标志>"
//       例：25.3℃ / 61.0% -> "@DATA ENV 253 610 1"
//       第三位为 0 表示 readDHT() 还没成功过，此时的 25/50 是上电默认值
//
// 为什么不能「收到就现场读 DHT」：
//   DHT11 一次读取要忙等 20~25ms，而下面 readDHT() 是连读两遍（湿度+温度）
//   ≈46ms；上位机的应答超时只有 300ms 且失败会重发 2 次。现场读会让多条
//   QENV 在 64 字节接收缓冲里排队，还会撞上重发时机。
//   所以这里只读 temperature / humidity 这两个已经被 readDHT() 更新好的全局量，
//   回执不到 1ms 就发完，完全不碰传感器时序、引脚和舵机/风扇逻辑。

// 64 字节留给 MSG 的文本，剩下的给命令字和参数
#define CMD_BUF_SIZE 72

char cmdBuf[CMD_BUF_SIZE];

uint8_t cmdLen=0;

// readDHT() 读到有效值时置 true；false 表示温湿度还是上电默认值
bool dhtOk=false;


//================ OLED 临时消息（新增） ================
//
// 契约：MSG <文本>  -> @ACK MSG <停留毫秒>
//       MSG          -> 立刻恢复常驻页，@ACK MSG 0
//
// 文本用 '|' 分隔，最多 3 行，黑底白字、整体垂直居中。
// 只认可打印 ASCII —— 6x12_tr 没有中文字模，发中文只会显示空白。
//
// 为什么必须自带超时：
//   OLED 的常驻职责是显示温湿度，告警页只是临时占用；上位机解除告警时
//   可能根本来不及发第二条命令（进程被杀、串口掉了），所以由板子自己
//   在 3 秒后回到常驻页，不依赖上位机的 CLEAR。

#define MSG_BUF_SIZE 64

// 消息停留时长：到期自动恢复常驻页
#define MSG_HOLD_MS 3000UL

char msgText[MSG_BUF_SIZE];

bool msgActive=false;

unsigned long msgUntil=0;


//================ 手动控制 FAN / LAMP（新增） ================
//
// 板子默认仍然自治：温度 >= 30 开风扇，天黑开灯。
//
// 控制权规则（固件与上位机约定）：
//   软件让设备「开」 -> 板子交出控制权并一直保持，controlFan/controlLamp 不再插手；
//   软件让设备「关」 -> 板子关掉后立刻交还自治，下一轮 loop 就按传感器重新判断。
//
// 为什么开的时候必须彻底锁住：
//   controlFan() 每个 loop 都会重写 FAN_PIN（30 开 / 28 关），controlLamp()
//   在天色翻转时也会重写灯带。不锁的话，手动指令活不过一轮 loop。
// 为什么关的时候必须立刻交还：
//   关掉之后设备不再影响传感器，自治判断重新变得可信（尤其灯：亮着时会把
//   自己的光敏照亮，只有在灯关掉之后，光敏读数才代表真实环境）。
//
// 协议：
//   FAN 1 / FAN 0           -> @ACK FAN <1|0>
//   LAMP 1 60 / LAMP 0      -> @ACK LAMP <1|0> <亮度 0~100>
//   LAMP AUTO               -> @ACK LAMP AUTO          （显式交还）
//   MSG ALERT|INTRUDER      -> @ACK MSG 3000           （OLED 临时页，3 秒后自动恢复）
//   MSG                     -> @ACK MSG 0              （立即恢复）
//   交还或自治改了状态      -> @EVT CTL AUTO <FAN|LAMP> / @EVT FAN ON|OFF / @EVT LAMP ON|OFF <亮度>

// 手动接管标志：true = 这个执行器归上位机管，自治逻辑不要动它
bool fanManual=false;
bool lampManual=false;

// 交还瞬间屏蔽「变暗 -> 自动开灯」的时长。
// 灯刚被手动关掉时，光敏会因为灯灭了而由亮转暗，这不是环境变化，
// 不挡住的话灯会在几十毫秒后自己又亮起来，表现为「关不掉」。
#define LAMP_SETTLE_MS 2000UL

unsigned long lampSettleUntil=0;

bool lampOn=false;

// 30 和原来 setup() 里的 strip.setBrightness(30) 对齐，手动开灯不会突然变亮
uint8_t lampLevel=30;

// 前端 0~100 映射到 NeoPixel 的 0~255 时封顶 100（约 39%）。
// 8 颗 WS2812 全白理论约 480mA，再叠上风扇会顶到 UNO 的 USB 500mA 自恢复保险丝
#define LED_LEVEL_MAX 100

// 已经通过 @ACK / @EVT 告诉过上位机的状态，用来判断还要不要补一条 @EVT
bool reportedFan=false;
bool reportedLamp=false;
uint8_t reportedLevel=30;


void applyFan(bool on)
{
  fanState=on;

  // 风扇模块是低电平触发：LOW 转，HIGH 停
  digitalWrite(FAN_PIN,on?LOW:HIGH);
}


void applyLamp(bool on,int level)
{
  // 先夹取再落到 uint8_t：直接 (uint8_t)300 会被截成 44
  if(level>100)
  {
    level=100;
  }

  if(level<0)
  {
    level=0;
  }

  lampOn=on;

  if(on&&level>0)
  {
    lampLevel=(uint8_t)level;
  }

  uint8_t b=0;

  if(lampOn)
  {
    b=(uint8_t)((uint16_t)lampLevel*LED_LEVEL_MAX/100);
  }

  strip.setBrightness(b);

  if(lampOn)
  {
    for(uint8_t i=0;i<LED_NUM;i++)
    {
      strip.setPixelColor(i,strip.Color(255,180,50));
    }
  }
  else
  {
    strip.clear();
  }

  strip.show();
}


// 显式交还灯：收到 LAMP AUTO 才走这里，按当前光态重算一次。
// 平时不用它——软件发 LAMP 0 时灯变 OFF 已经等于交还了。
void releaseLamp()
{
  if(!lampManual)
  {
    return;
  }

  lampManual=false;

  applyLamp(darkState,lampLevel);

  oldDarkState=darkState;

  Serial.println(F("@EVT CTL AUTO LAMP"));
}


// 状态变了就补报一行，页面才不会和板子各说各话
void reportStates()
{
  if(fanState!=reportedFan)
  {
    reportedFan=fanState;

    Serial.print(F("@EVT FAN "));

    if(fanState)
    {
      Serial.println(F("ON"));
    }
    else
    {
      Serial.println(F("OFF"));
    }
  }


  if(lampOn!=reportedLamp||(lampOn&&lampLevel!=reportedLevel))
  {
    reportedLamp=lampOn;

    reportedLevel=lampLevel;

    Serial.print(F("@EVT LAMP "));

    if(lampOn)
    {
      Serial.print(F("ON "));
    }
    else
    {
      Serial.print(F("OFF "));
    }

    Serial.println(lampOn?lampLevel:0);
  }
}


// 命令执行完先同步一次「已上报」，免得紧跟着再发一条一模一样的 @EVT
void syncReported()
{
  reportedFan=fanState;

  reportedLamp=lampOn;

  reportedLevel=lampLevel;
}


// 取命令里第 idx 个空格分隔参数（从 1 开始），没有就返回 def
int cmdArg(const char* cmd,uint8_t idx,int def)
{
  uint8_t n=0;

  while(*cmd&&n<idx)
  {
    if(*cmd==' ')
    {
      n++;

      if(n==idx)
      {
        cmd++;

        break;
      }
    }

    cmd++;
  }

  if(n<idx||*cmd=='\0')
  {
    return def;
  }

  return atoi(cmd);
}


// 把缓存值按协议回给上位机（整数、放大 10 倍，避免 AVR 上打印浮点）
void replyEnv()
{
  Serial.print(F("@DATA ENV "));

  Serial.print((long)(temperature*10.0+0.5));

  Serial.print(' ');

  Serial.print((long)(humidity*10.0+0.5));

  Serial.print(' ');

  Serial.println(dhtOk?1:0);
}


// 只认全大写的命令字，避免把复位瞬间的残留字节当成命令去 NAK
bool isCmdToken(const char* s)
{
  if(s[0]=='\0')
  {
    return false;
  }

  for(uint8_t i=0;s[i];i++)
  {
    if(s[i]<'A'||s[i]>'Z')
    {
      return false;
    }
  }

  return true;
}


void handleCmd(const char* cmd)
{
  if(!strcmp(cmd,"QENV"))
  {
    replyEnv();
  }

  else if(!strcmp(cmd,"PING"))
  {
    // 心跳：上位机靠它判断环境板是否在线
    Serial.print(F("@PONG "));

    Serial.println(millis());
  }

  else if(!strncmp(cmd,"FAN",3)&&(cmd[3]=='\0'||cmd[3]==' '))
  {
    int on=cmdArg(cmd,1,-1);

    if(on<0||on>1)
    {
      Serial.println(F("@NAK FAN"));

      return;
    }

    // 开：接管，一直保持；关：先关掉，再立刻交还自治
    fanManual=(on==1);

    applyFan(on==1);

    syncReported();

    Serial.print(F("@ACK FAN "));

    Serial.println(on);

    if(on==0)
    {
      // 本轮稍后的 controlFan() 就会按温度重新判断：
      // 温度够高它会自己转起来，并通过 @EVT FAN ON 告诉上位机
      Serial.println(F("@EVT CTL AUTO FAN"));
    }
  }

  else if(!strncmp(cmd,"LAMP",4)&&(cmd[4]=='\0'||cmd[4]==' '))
  {
    // LAMP AUTO：显式交还，交回光敏自治
    if(!strncmp(cmd+4," AUTO",5))
    {
      releaseLamp();

      Serial.println(F("@ACK LAMP AUTO"));

      return;
    }

    int on=cmdArg(cmd,1,-1);

    // 第二个参数是亮度 0~100，省略就沿用当前值
    int lv=cmdArg(cmd,2,-1);

    if(on<0||on>1)
    {
      Serial.println(F("@NAK LAMP"));

      return;
    }

    // 开：接管，一直保持；关：灭灯即交还
    lampManual=(on==1);

    applyLamp(on==1,(lv>=0)?lv:(int)lampLevel);

    if(!lampOn)
    {
      // 灯灭会让光敏由亮转暗，这不是环境变化，先挡住，别让它自己又亮起来
      lampSettleUntil=millis()+LAMP_SETTLE_MS;
    }

    syncReported();

    Serial.print(F("@ACK LAMP "));

    Serial.print(lampOn?1:0);

    Serial.print(' ');

    Serial.println(lampLevel);

    if(!lampOn)
    {
      Serial.println(F("@EVT CTL AUTO LAMP"));
    }
  }

  else if(!strncmp(cmd,"MSG",3)&&(cmd[3]=='\0'||cmd[3]==' '))
  {
    const char* text=cmd+3;

    while(*text==' ')
    {
      text++;
    }

    // 空文本 = 提前恢复，给上位机留一个显式撤销的口子
    if(*text=='\0')
    {
      msgActive=false;

      displayOLED();

      Serial.println(F("@ACK MSG 0"));

      return;
    }

    // 截断也要留结尾 0：strncpy 不补 '\0'，直接打印会越界读后面内存
    strncpy(msgText,text,MSG_BUF_SIZE-1);

    msgText[MSG_BUF_SIZE-1]='\0';

    msgUntil=millis()+MSG_HOLD_MS;

    msgActive=true;

    // 立刻画一帧：等 2 秒的常驻刷新再画，3 秒停留就只剩 1 秒了
    displayOLED();

    Serial.print(F("@ACK MSG "));

    Serial.println(MSG_HOLD_MS);
  }

  else if(isCmdToken(cmd))
  {
    Serial.print(F("@NAK "));

    Serial.println(cmd);
  }

  // 其余一律当噪声丢弃
}


// 非阻塞收行：逐字符攒到 '\n' 再整行分派。
// 千万不能用「攒满一行再整体比较」的老写法，多条命令连发时会拼成一串。
void pollSerial()
{
  while(Serial.available()>0)
  {
    char c=(char)Serial.read();

    if(c=='\n'||c=='\r')
    {
      if(cmdLen>0)
      {
        cmdBuf[cmdLen]='\0';

        handleCmd(cmdBuf);

        cmdLen=0;
      }
    }

    else if(cmdLen<CMD_BUF_SIZE-1)
    {
      cmdBuf[cmdLen++]=c;
    }

    else
    {
      // 超长行直接丢弃，防止缓冲区溢出
      cmdLen=0;
    }
  }
}


//================ SETUP ================


void setup()
{

Serial.begin(9600);


// SPI保护

pinMode(10,OUTPUT);

digitalWrite(10,HIGH);


// FAN

pinMode(FAN_PIN,OUTPUT);

digitalWrite(FAN_PIN,HIGH);


// WS2812

strip.begin();

strip.setBrightness(30);

strip.clear();

strip.show();


// 光敏

pinMode(LDR_PIN,INPUT);


// OLED

oled.begin();

oled.setContrast(120);


oled.clearBuffer();

oled.setFont(
u8g2_font_6x12_tr
);


oled.drawStr(
0,
20,
"Smart Home"
);


oled.drawStr(
0,
40,
"Starting..."
);


oled.sendBuffer();



// DHT

dht.begin();


delay(1000);


// 启动横幅：@BOOT 1 让上位机知道这块板支持串口命令（1 = 协议版本 1）。
// 老固件打的是裸文本 "Ready"，上位机会当 legacy 处理并改用 QRAIN 保活，
// 而环境板并没有雨滴模块，链路会被反复误判成离线。
Serial.println(F("@BOOT 1"));


}






//================ LOOP =================


void loop()
{

// 先收命令再做别的：回执只是拼一行字符串，不阻塞，也不影响下面各段时序
pollSerial();


// 接管与交还都在 pollSerial 里按命令完成，这里不再需要额外动作

readLight();



if(millis()-dhtTimer>2000)
{

readDHT();

dhtTimer=millis();

}



controlFan();


controlLamp();


// 自治改了风扇/灯带就补一条 @EVT，让页面跟上
reportStates();



// 有临时消息时 200ms 一帧（倒计时条才是连续的），常驻页 2s 一帧足够
unsigned long oledPeriod=msgActive?200UL:2000UL;

if(millis()-oledTimer>oledPeriod)
{

displayOLED();

oledTimer=millis();

}


}
//================ DHT读取 =================

void readDHT()
{

  float h=dht.readHumidity();

  float t=dht.readTemperature();


  if(!isnan(h)&&!isnan(t))
  {

    humidity=h;

    temperature=t;

    dhtOk=true;   // [QENV] 只加这一个标志，采样时序不变

  }


}



//================ 光敏读取 =================

void readLight()
{

int value=digitalRead(LDR_PIN);


bool nowDark;


if(value==HIGH)
{
nowDark=true;
}
else
{
nowDark=false;
}



if(nowDark==lastLight)
{
lightCount++;
}
else
{
lightCount=0;
}


lastLight=nowDark;



if(lightCount>10)
{
darkState=nowDark;
}


}

//================ 风扇控制 =================


void controlFan()
{

  // 上位机接管期间（FAN 1 之后）不碰引脚，
  // 否则刚发的 FAN 1 会被这里按 30/28 立刻改回去
  if(fanManual)
  {
    return;
  }

  if(temperature >= FAN_ON_TEMP)
  {

    fanState=true;

    digitalWrite(
      FAN_PIN,
      LOW
    );

  }


  else if(temperature <= FAN_OFF_TEMP)
  {

    fanState=false;

    digitalWrite(
      FAN_PIN,
      HIGH
    );

  }

}



//================ 灯带控制 =================


void controlLamp()
{

// 上位机接管期间（LAMP 1 之后）不接管
if(lampManual)
{
return;
}


if(darkState!=oldDarkState)
{


oldDarkState=darkState;


// 刚把手动关掉的灯交还自治时，光敏会因为灯灭了而由亮转暗。
// 这不是环境变化，2 秒内忽略「变暗」这一侧，否则灯会自己又亮起来
if(darkState&&(long)(millis()-lampSettleUntil)<0)
{
return;
}


// 交给 applyLamp 统一处理：亮度和状态只有一份，OLED 与上报都读得到
applyLamp(darkState,lampLevel);


}


}





//================ OLED显示 =================


// 常驻页：温湿度 + 风扇。只在没有临时消息时出现
void drawStatus()
{

oled.setFont(u8g2_font_6x12_tr);

oled.drawStr(
0,
12,
"SMART HOME"
);


oled.setCursor(0,30);
oled.print("TEMP:");
oled.print(temperature);
oled.print(" C");


oled.setCursor(0,45);
oled.print("HUM:");
oled.print(humidity);
oled.print(" %");


oled.setCursor(0,60);
oled.print("FAN:");

if(fanState)
oled.print("ON");
else
oled.print("OFF");

}


// 临时消息页：黑底白字，最多 3 行，整体垂直居中，底部一条细的剩余时间条。
// 刻意不做反白标题块——满屏白块要点亮上千个像素，既费电又容易留残影。
void drawMsg()
{

long remain=(long)(msgUntil-millis());

if(remain<0)
{
remain=0;
}


oled.setFont(u8g2_font_6x12_tr);


// 先数行数，再定起始基线：1 行居中 36，2 行 28/44，3 行 20/36/52
uint8_t lines=1;

for(const char* q=msgText;*q;q++)
{
if(*q=='|')
{
lines++;
}
}

if(lines>3)
{
lines=3;
}


uint8_t y=(uint8_t)(36-((lines-1)*8));


const char* p=msgText;

uint8_t line=0;

while(*p&&line<3)
{

char buf[22];

uint8_t n=0;

while(*p&&*p!='|'&&n<21)
{
buf[n++]=*p++;
}

buf[n]='\0';

if(*p=='|')
{
p++;
}


uint8_t w=(uint8_t)oled.getStrWidth(buf);

oled.drawStr((w<128)?(uint8_t)((128-w)/2):0,y,buf);


y+=16;

line++;

if(!*p)
{
break;
}

}


// 3 秒倒计时条：1px 细框 + 3px 填充，亮起的像素约 350 个，
// 比原来那条 128x15 的反白块（约 1920 个）少一个量级
uint8_t bar=(uint8_t)((remain*116L)/MSG_HOLD_MS);

oled.drawFrame(4,57,120,5);

if(bar>0)
{
oled.drawBox(5,58,(bar>116)?116:bar,3);
}

}


// 分派：临时消息到期就地翻回常驻页，不需要上位机再发命令
void displayOLED()
{

if(msgActive&&(long)(millis()-msgUntil)>=0)
{
msgActive=false;
}


oled.firstPage();

do
{

if(msgActive)
{
drawMsg();
}

else
{
drawStatus();
}

}
while(oled.nextPage());

}
